<calcdefinition dosierungswechsel>
title = Dosierung wechseln – Träger als Rest
showwork = true

input gesamtmenge:
  type = decimal
  label = Gesamtmenge der Rezeptur einschließlich Träger
  unit = kg
  default = 1
  min = 0.001

input dosierung_alt:
  type = decimal
  label = Bisherige Dosierung
  unit = %
  default = 2
  min = 0.001
  max = 100

input dosierung_neu:
  type = decimal
  label = Neue Dosierung
  unit = %
  default = 3
  min = 0.001
  max = 100

input mengen:
  type = regex-list
  label = Komponenten ohne Träger – eine kg-Menge je Zeile
  default = "Artikel	Menge
ITEM-A	0,500 kg
ITEM-B	0,060 kg
ITEM-C	0,040 kg"
  pattern = (?<value>[+-]?\d+(?:[.,]\d+)?)(?=\s+kg(?:\s|$))
  flags = iu
  min = 0
  break = true

output komponenten_neu:
  type = text-list
  source = mengen
  label = Neue Komponentenmengen ohne Träger
  formula = value * dosierung_alt / dosierung_neu
  unit = kg
  decimals = 3
  fixed = true
  break = true

output komponentensumme:
  label = Summe der neuen Komponenten ohne Träger
  unit = kg
  formula = sum(komponenten_neu)
  decimals = 3
  fixed = true

output traeger_neu:
  label = Neuer Trägerrest – negative Werte sind nicht verwendbar
  unit = kg
  formula = gesamtmenge - komponentensumme
  decimals = 3
  fixed = true

output anzahl:
  label = Anzahl erkannter Komponenten – bitte prüfen
  formula = count(mengen)
  decimals = 0
</calcdefinition>
