<calcdefinition dilution>
title = Dilution
showwork = true

input concentration:
  type = decimal
  label = Initial concentration
  unit = %
  default = 40
  min = 0
  max = 100

input addition:
  type = decimal
  label = Fraction of the initial solution in the final mixture
  unit = %
  default = 2
  min = 0
  max = 100

output result:
  label = Final concentration
  unit = %
  formula = concentration * addition / 100
  decimals = 3
  fixed = true
</calcdefinition>
