====== Rechnerdefinitionen ======

<calcdefinition rezept>
title = Rezeptmengen umrechnen

input personen_alt:
  type = decimal
  label = Bisherige Personenanzahl
  default = 2
  min = 1

input personen_neu:
  type = decimal
  label = Gewünschte Personenanzahl
  default = 5
  min = 1

input mengen:
  break = true
  type = decimal-list
  label = Mengen – eine Zahl pro Zeile

output ergebnis:
  label = Umgerechnete Mengen
  source = mengen
  formula = value * personen_neu / personen_alt
  decimals = 3
</calcdefinition>
