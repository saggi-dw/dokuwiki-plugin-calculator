<calcdefinition rezept_text>
title = Rezeptmengen mit Text umrechnen

input personen_alt:
  type = decimal
  label = Bisherige Personenanzahl
  default = 2
  min = 1

input personen_neu:
  type = decimal
  label = Gewünschte Personenanzahl
  default = 5
  min = 1

input mengen:
  break = true
  type = regex-list
  label = Zutaten – eine Zeile pro Zutat
  pattern = ^\s*(?:[•*]\s*|-\s+)?(?<value>[+-]?(?:\d+(?:[.,]\d+)?|[.,]\d+))
  flags = u

output ergebnis:
  type = text-list
  label = Umgerechnete Zutaten
  source = mengen
  formula = value * personen_neu / personen_alt
  decimals = 3
</calcdefinition>
