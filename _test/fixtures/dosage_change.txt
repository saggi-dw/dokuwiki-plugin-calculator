<calcdefinition dosage_change>
title = Change dosage – carrier remainder
showwork = true

input batch_quantity:
  type = decimal
  label = Total batch quantity including carrier
  unit = kg
  default = 1
  min = 0.001

input original_dosage:
  type = decimal
  label = Original dosage
  unit = %
  default = 2
  min = 0.001
  max = 100

input target_dosage:
  type = decimal
  label = Target dosage
  unit = %
  default = 3
  min = 0.001
  max = 100

input quantities:
  type = regex-list
  label = Components without carrier – one kg quantity per line
  default = "Product	Quantity
ITEM-A	0,500 kg
ITEM-B	0,060 kg
ITEM-C	0,040 kg"
  pattern = (?<value>[+-]?\d+(?:[.,]\d+)?)(?=\s+kg(?:\s|$))
  flags = iu
  min = 0
  break = true

output scaled_components:
  type = text-list
  source = quantities
  label = Scaled component quantities without carrier
  formula = value * original_dosage / target_dosage
  unit = kg
  decimals = 3
  fixed = true
  break = true

output component_total:
  label = Total scaled quantity without carrier
  unit = kg
  formula = sum(scaled_components)
  decimals = 3
  fixed = true

output carrier_remainder:
  label = Carrier remainder – negative values are not usable
  unit = kg
  formula = batch_quantity - component_total
  decimals = 3
  fixed = true

output item_count:
  label = Recognized component count – please check
  formula = count(quantities)
  decimals = 0
</calcdefinition>
