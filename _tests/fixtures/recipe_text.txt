<calcdefinition recipe_text>
title = Scale recipe text

input original_servings:
  type = decimal
  label = Original servings
  default = 2
  min = 1

input target_servings:
  type = decimal
  label = Target servings
  default = 5
  min = 1

input quantities:
  break = true
  type = regex-list
  label = Ingredients – one ingredient per line
  pattern = ^\s*(?:[•*]\s*|-\s+)?(?<value>[+-]?(?:\d+(?:[.,]\d+)?|[.,]\d+))
  flags = u

output result:
  type = text-list
  label = Scaled ingredients
  source = quantities
  formula = value * target_servings / original_servings
  decimals = 3
</calcdefinition>
